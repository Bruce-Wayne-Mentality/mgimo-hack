## 🧮 Сводка выполнения (опционально: Dataview)

> [!note] Один раз включите Dataview → Enable JavaScript Queries
> Без Dataview этот блок будет показываться как исходный код. Вся остальная система продолжит работать.

```dataviewjs
const tasks = [];
for (const page of dv.pages('"01 — Разработка"')) {
  for (const task of page.file.tasks) {
    if (task.text.includes('#dekanat')) tasks.push(task);
  }
}
const values = [
  ['Все задачи', tasks],
  ['P0', tasks.filter(t => t.text.includes('#p0'))],
  ['P1', tasks.filter(t => t.text.includes('#p1'))],
  ['P2', tasks.filter(t => t.text.includes('#p2'))],
  ['Roadmap', tasks.filter(t => t.text.includes('#roadmap'))]
];
dv.table(['Срез', 'Готово', 'Всего', 'Прогресс'], values.map(([name, items]) => {
  const done = items.filter(t => t.completed).length;
  const total = items.length;
  return [name, done, total, total ? `${Math.round(100*done/total)}%` : '—'];
}));
```
